        function formatThaiDateShort(date) {
            if (!date) return '-';
            const d = new Date(date);
            const day = d.getDate();
            const monthNames = ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'];
            const month = monthNames[d.getMonth()];
            const year = (d.getFullYear() + 543).toString().slice(-2);
            return `${day} ${month} ${year}`;
        }

        function padCode(code) {

        function loadOptions() {
            showLoading(true);
            return new Promise((resolve, reject) => {
                google.script.run
                    .withSuccessHandler((result) => {
                        showLoading(false);
                        if (result && result.success) {
                            optionsData = result;
                            populateOptions();
                            document.querySelectorAll('.equipment-row').forEach(row => setupEquipmentRow(row));
                            resolve();
                        } else {
                            Swal.fire({
                                title: 'ข้อผิดพลาด',
                                text: result ? result.message : 'ไม่สามารถโหลดตัวเลือกได้ กรุณาตรวจสอบสเปรดชีต',
                                icon: 'error',
                                customClass: {
                                    confirmButton: 'btn btn-submit'
                                }
                            });
                            reject(result ? result.message : 'Unknown error');
                        }
                    })
                    .withFailureHandler((error) => {
                        showLoading(false);
                        Swal.fire({
                            title: 'ข้อผิดพลาด',
                            text: 'เกิดข้อผิดพลาดในการโหลดตัวเลือก: ' + error.message,
                            icon: 'error',
                            customClass: {
                                confirmButton: 'btn btn-submit'
                            }
                        });
                        reject(error.message);
                    })
                    .getOptions();
            });
        }
